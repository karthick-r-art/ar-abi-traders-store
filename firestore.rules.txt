// Firestore security rules for A.R. Abi Traders order + product sync.
// Paste this into: Firebase Console → Firestore Database → Rules tab → Publish.
//
// orders/{orderId} — what this allows:
//  - Anyone (customers, no login) can CREATE a new order — needed so the
//    storefront can save an order without asking shoppers to sign in.
//  - Anyone can READ a single order if they know its exact order number
//    (used by the live tracker + "Track my order" page). Nobody can list
//    or browse all orders without being logged in.
//  - Only the logged-in shop owner (Firebase Authentication) can UPDATE an
//    order (e.g. change status to "confirmed") or read the full order list
//    (used by admin.html).
//  - Nobody can delete orders.
//
// productOverrides/{productId} — price/stock/availability edits made in
// admin.html. Anyone can read the whole collection (needed so the public
// storefront shows current prices), but only the logged-in shop owner can
// write to it.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /orders/{orderId} {
      allow get: if true;
      allow list: if request.auth != null;
      allow create: if request.resource.data.num == orderId
                    && request.resource.data.status == "received";
      allow update: if request.auth != null;
      allow delete: if false;
    }
    match /productOverrides/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
