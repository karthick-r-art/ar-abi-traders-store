// Firebase Storage security rules for A.R. Abi Traders product photos.
// Paste this into: Firebase Console → Storage → Rules tab → Publish.
// (This is a different product from Firestore, with its own separate Rules tab.)
//
// What this allows:
//  - Anyone can VIEW product photos (needed so the public storefront can show them).
//  - Only the logged-in shop owner (Firebase Authentication) can UPLOAD a photo.
//  - Uploads must be an image file under 5MB (blocks accidental huge files / abuse).

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /products/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }
  }
}
